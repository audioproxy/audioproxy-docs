import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Describe the variant in the URL" icon="pencil">
    `f:opus/br:96/t:0:30/fade:1:1` is a 30-second faded Opus preview. Options
    double as the cache key, so one URL always means the same bytes. Sign it
    once and ship it anywhere.
  </Card>
  <Card title="Stream before the encode finishes" icon="rocket">
    First bytes reach the client while ffmpeg is still working. Cached
    variants serve with `Content-Length` and Range support, so players seek
    instantly on every play after the first.
  </Card>
  <Card title="Deploy one container" icon="seti:docker">
    Sources on a mounted directory or in any S3-compatible store; the variant
    cache likewise. No database, no queue, no sidecar; configuration is a
    handful of environment variables.
  </Card>
  <Card title="Build against the contract" icon="open-book">
    The [API contract](/reference/api-v1/) fixes the URL grammar, cache-key
    rules and error codes; module-level reference lives on
    [hexdocs.pm/audio_proxy](https://hexdocs.pm/audio_proxy). This site covers
    operating and integrating.
  </Card>
</CardGrid>
